import SwiftUI
import DesignKitIOS

/// 独立可复制示例：所有层级、角色、忙碌与禁用。
struct ButtonExample: View {
    let isBusy: Bool
    @State private var count = 0

    init(isBusy: Bool = false) { self.isBusy = isBusy }

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            EButton("Primary action", emphasis: .primary, isBusy: isBusy) { count += 1 }
            EButton("Secondary action") { count += 1 }
            EButton("Tertiary action", emphasis: .tertiary) { count += 1 }
            EButton("Destructive action", role: .destructive) { count += 1 }
            EButton("Disabled action") {}.disabled(true)
            Text(verbatim: "Action count: \(count)")
        }
    }
}
